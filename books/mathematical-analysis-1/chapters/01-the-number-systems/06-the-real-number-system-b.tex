\begin{theorem} \label{an1:thm:existence-and-uniqueness-of-nth-roots}
  \[
    \forall x \in \R^{+}, \forall n \in \N, \exists! y \in \R^{+}, y^{n} = x
  \]
\end{theorem}

\begin{proof}
  Suppose
  \begin{itemize}
    \item $x \in \R^{+}$
    \item $n \in \N$
  \end{itemize}
  Define
  \[
    E \coloneqq \setbuilder*{t \in \R^{+}}{t^{n} < x} \subseteq \R^{+}
  \]
  Then
  \begin{itemize}
    \item If $x \geq 1$, then
          \[
            \begin{array}{c}
              2^{-n} < 1 \leq x \\
              \Downarrow        \\
              2^{-1} \in E
            \end{array}
          \]
          Also,
          \[
            \begin{array}{c}
              \forall t \in E, t^{n} < x \leq x^{n}                \\
              \by{\Cref{an1:lem:nth-power-is-monotone}} \Downarrow \\
              \forall t \in E, t < x                               \\
              \Downarrow                                           \\
              x \in E^{u}
            \end{array}
          \]
    \item Otherwise,
          \[
            \begin{array}{c}
              x < 1      \\
              \Downarrow \\
              x^{2n} < x \\
              \Downarrow \\
              x^{2} \in E
            \end{array}
          \]
          Also,
          \[
            \begin{array}{c}
              \forall t \in E, t^{n} < x < 1 = 1^{n}               \\
              \by{\Cref{an1:lem:nth-power-is-monotone}} \Downarrow \\
              \forall t \in E, t < 1                               \\
              \Downarrow                                           \\
              1 \in E^{u}
            \end{array}
          \]
  \end{itemize}
  Therefore,
  \[
    \begin{array}{c}
      E \neq \vnot \land E^{u} \neq \vnot        \\
      \by{\Cref{an1:ax:completeness}} \Downarrow \\
      \exists \alpha \in \R^{+}, \alpha = \sup E
    \end{array}
  \]
  We will show
  \[
    \begin{array}{c}
      \alpha^{n} \not< x \land \alpha^{n} \not> x \\
      \Downarrow                                  \\
      \alpha^{n} = x
    \end{array}
  \]
  \begin{itemize}
    \item Assume $\alpha^{n} < x$. By \cref{an1:thm:Q-is-dense-in-R},
          \[
            \begin{array}{c}
              \exists h \in \Q, 0 < h < \min\set*{1, \frac{x - \alpha^{n}}{n(\alpha + 1)^{n - 1}}}       \\
              \by{\Cref{an1:lem:nth-power-is-lipschitz}} \Downarrow                                      \\
              (\alpha + h)^{n} - \alpha^{n} \leq hn(\alpha + h)^{n - 1} < hn(\alpha + 1)^{n - 1}         \\
              \Downarrow                                                                                 \\
              (\alpha + h)^{n} < \alpha^{n} + hn(\alpha + 1)^{n - 1} < \alpha^{n} + (x - \alpha^{n}) = x \\
              \Downarrow                                                                                 \\
              \alpha + h \in E                                                                           \\
              \by{\ensuremath{\alpha = \sup E}} \Downarrow                                               \\
              \bot
            \end{array}
          \]
    \item Assume $\alpha^{n} > x$. By \cref{an1:thm:Q-is-dense-in-R},
          \[
            \begin{array}{c}
              \exists h \in \Q, 0 < h < \min\set*{\alpha, \frac{\alpha^{n} - x}{n\alpha^{n - 1}}}     \\
              \by{\Cref{an1:lem:nth-power-is-lipschitz}} \Downarrow                                   \\
              \alpha^{n} - (\alpha - h)^{n} \leq hn\alpha^{n - 1}                                     \\
              \Downarrow                                                                              \\
              (\alpha - h)^{n} \geq \alpha^{n} - hn\alpha^{n - 1} > \alpha^{n} - (\alpha^{n} - x) = x \\
              \Downarrow                                                                              \\
              \forall t \in E, t^{n} < x < (\alpha - h)^{n}                                           \\
              \by{\Cref{an1:lem:nth-power-is-monotone}} \Downarrow                                    \\
              \forall t \in E, t < \alpha - h                                                         \\
              \Downarrow                                                                              \\
              \alpha - h \in E^{u}                                                                    \\
              \by{\ensuremath{\alpha = \sup E}} \Downarrow                                            \\
              \bot
            \end{array}
          \]
  \end{itemize}
  This implies
  \[
    \exists y \in \R^{+}, y^{n} = x
  \]
  Suppose
  \begin{itemize}
    \item $y_{1}, y_{2} \in \R^{+}$
    \item $\paren*{y_{1}}^{n} = \paren*{y_{2}}^{n} = x$
  \end{itemize}
  Assume $y_{1} \neq y_{2}$ (\emph{WLOG} $y_{1} < y_{2}$). Then
  \[
    \begin{array}{c}
      y_{1} < y_{2}                                        \\
      \by{\Cref{an1:lem:nth-power-is-monotone}} \Downarrow \\
      \paren*{y_{1}}^{n} < \paren*{y_{2}}^{n}              \\
      \Downarrow                                           \\
      \bot
    \end{array}
  \]
  Therefore,
  \[
    y_{1} = y_{2}
  \]
  This implies
  \[
    \exists! y \in \R^{+}, y^{n} = x
  \]
\end{proof}

\begin{corollary} \label{an1:cor:existence-and-uniqueness-of-nth-roots}
  \[
    \forall x \in [0, \infty), \forall n \in \N, \exists! y \in [0, \infty), y^{n} = x
  \]
  Such an $y$ is denoted by
  \[
    y = \sqrt[n]{x} = x^{\frac{1}{n}}
  \]
\end{corollary}

\begin{proof}
  Suppose
  \begin{itemize}
    \item $x \in [0, \infty)$
    \item $n \in \N$
  \end{itemize}
  Then
  \begin{itemize}
    \item If $x = 0$, then
          \begin{itemize}
            \item $0^{n} = 0$
            \item $\forall y \in [0, \infty), y \neq 0 \implies y^{n} \neq 0$
          \end{itemize}
    \item Otherwise,
          \[
            \begin{array}{c}
              x \in \R^{+}                                                         \\
              \by{\Cref{an1:thm:existence-and-uniqueness-of-nth-roots}} \Downarrow \\
              \exists! y \in \R^{+}, y^{n} = x
            \end{array}
          \]
  \end{itemize}
\end{proof}

\begin{definition} \label{an1:def:greedy-base-expansion}
  Suppose
  \begin{itemize}
    \item $y \in [0, 1)$
    \item $b \in [2, \infty)_{\N}$
  \end{itemize}
  A sequence $\paren*{a_{n}}_{n = 1}^{\infty}$ in $[0, b - 1]_{\N}$ is called a \textbf{greedy base $b$ expansion} of $y$ if
  \[
    \forall n \in \N, 0 \leq y - \sum_{k = 1}^{n} \frac{a_{k}}{b^{k}} < \frac{1}{b^{n}}
  \]
  Such an $\paren*{a_{n}}_{n = 1}^{\infty}$ is denoted by
  \[
    \paren*{a_{n}}_{n = 1}^{\infty} = \mathcal{G}_{b}(y)
  \]
\end{definition}

\begin{theorem} \label{an1:thm:greedy-base-expansion-exists-uniquely}
  \[
    \forall b \in [2, \infty)_{\N}, \forall y \in [0, 1), \exists! \paren*{a_{n}}_{n = 1}^{\infty}
    \in \prod_{n = 1}^{\infty} [0, b - 1]_{\N}, \paren*{a_{n}}_{n = 1}^{\infty} = \mathcal{G}_{b}(y)
  \]
\end{theorem}
