\begin{corollary} \label{an:cor:infinite-subset-implies-infinite}
  \[
    \forall X \forall Y, \abs*{Y} \geq \aleph_{0} \land Y \subseteq X \implies \abs*{X} \geq \aleph_{0}
  \]
\end{corollary}

\begin{proof}
  Suppose
  \begin{itemize}
    \item $\abs*{Y} \geq \aleph_{0}$
    \item $Y \subseteq X$
  \end{itemize}
  Assume
  \[
    \abs*{X} < \aleph_{0}
  \]
  By \cref{an:thm:subset-of-finite-set-is-finite},
  \[
    \begin{array}{c}
      \abs*{Y} < \aleph_{0} \\
      \Downarrow            \\
      \bot
    \end{array}
  \]
  Therefore,
  \[
    \abs*{X} \geq \aleph_{0}
  \]
\end{proof}

\begin{lemma} \label{an:lem:surjection-from-Nn-implies-finite}
  \[
    \forall Y, \paren*{\exists n \in [0, \infty)_{\N}, \exists f: [1, n]_{\N} \twoheadrightarrow Y}
    \implies \abs*{Y} < \aleph_{0}
  \]
\end{lemma}

\begin{proof}
  Suppose
  \[
    \exists n \in [0, \infty)_{\N}, \exists f: [1, n]_{\N} \twoheadrightarrow Y
  \]
  Define
  \[
    \forall y \in Y, N_{y} \coloneqq f^{-1}(\set*{y}) \subseteq [1, n]_{\N}
  \]
  Then
  \[
    \begin{array}{c}
      \forall y \in Y, N_{y} \neq \vnot                    \\
      \by{\Cref{an:thm:well-ordering-property}} \Downarrow \\
      \forall y \in Y, \exists n_{y} \in N_{y}, n_{y} = \min N_{y}
    \end{array}
  \]
  Define
  \[
    M \coloneqq \setbuilder*{n_{y}}{y \in Y} \subseteq [1, n]_{\N}
  \]
  Define $g: M \to Y$ by
  \[
    g \equiv f|_{M}^{Y}
  \]
  Then
  \begin{itemize}
    \item \textbf{Injectivity}: Suppose
          \begin{itemize}
            \item $n_{y_{1}}, n_{y_{2}} \in M$
            \item $g(n_{y_{1}}) = g(n_{y_{2}})$
          \end{itemize}
          Then
          \[
            \begin{array}{c}
              f(n_{y_{1}}) = f(n_{y_{2}}) \\
              \Downarrow                  \\
              y_{1} = y_{2}               \\
              \Downarrow                  \\
              N_{y_{1}} = N_{y_{2}}       \\
              \Downarrow                  \\
              n_{y_{1}} = n_{y_{2}}
            \end{array}
          \]
    \item \textbf{Surjectivity}: Suppose $y \in Y$. Then
          \[
            \begin{array}{c}
              N_{y} \neq \vnot                                     \\
              \by{\Cref{an:thm:well-ordering-property}} \Downarrow \\
              \exists n_{y} \in N_{y}, n_{y} = \min N_{y}          \\
              \Downarrow                                           \\
              g(n_{y}) = y
            \end{array}
          \]
  \end{itemize}
  Therefore,
  \[
    g: M \xrightarrow{\sim} Y
  \]
  By \cref{an:lem:subset-of-Nn-is-finite},
  \[
    \begin{array}{c}
      \abs*{M} < \aleph_{0}                                        \\
      \Downarrow                                                   \\
      \exists m \in [0, \infty)_{\N}, [1, m]_{\N} \sim M           \\
      \by{\Cref{an:thm:bijection-equivalence-relation}} \Downarrow \\
      \abs*{Y} = m < \aleph_{0}
    \end{array}
  \]
\end{proof}

\begin{theorem} \label{an:thm:surjection-from-finite-set-implies-finite}
  \[
    \forall X \forall Y, \abs*{X} < \aleph_{0} \land \exists f: X \twoheadrightarrow Y
    \implies \abs*{Y} < \aleph_{0}
  \]
\end{theorem}

\begin{proof}
  Suppose
  \begin{itemize}
    \item $\abs*{X} < \aleph_{0}$
    \item $\exists f: X \twoheadrightarrow Y$
  \end{itemize}
  Then
  \[
    \begin{array}{c}
      \exists n \in [0, \infty)_{\N}, \exists g: [1, n]_{\N} \xrightarrow{\sim} X \\
      \by{\Cref{an:thm:properties-of-composition-functions}} \Downarrow           \\
      f \circ g: [1, n]_{\N} \twoheadrightarrow Y                                 \\
      \by{\Cref{an:lem:surjection-from-Nn-implies-finite}} \Downarrow             \\
      \abs*{Y} < \aleph_{0}
    \end{array}
  \]
\end{proof}

\begin{theorem} \label{an:thm:finite-ordered-set-has-extremum}
  Suppose $X$ is ordered. Then
  \[
    0 < \abs*{X} < \aleph_{0} \implies \exists m, M \in X, m = \min X \land M = \max X
  \]
\end{theorem}


\begin{proof}
  Define
  \[
    \forall n \in \N, P(n): \forall X, \abs*{X} = n \implies \exists m, M \in X, m = \min X \land M = \max X
  \]
  Then
  \begin{itemize}
    \item \textbf{Base case:} Suppose $X = \set*{a_{1}}$. Then
          \begin{itemize}
            \item $\forall x \in X, a_{1} \leq x$
            \item $\forall x \in X, x \leq a_{1}$
          \end{itemize}
          Therefore, $P(1)$ is true.
    \item \textbf{Inductive step:} Suppose $P(n)$ is true. Suppose
          \begin{itemize}
            \item $X = \set*{a_{1}, a_{2}, \dots, a_{n + 1}}$
            \item $Y = \set*{a_{1}, a_{2}, \dots, a_{n}} \subseteq X$
          \end{itemize}
          Since $P(n)$ is true,
          \begin{itemize}
            \item $\exists k \in [1, n]_{\N}, \forall y \in Y, a_{k} \leq y$
            \item $\exists K \in [1, n]_{\N}, \forall y \in Y, y \leq a_{K}$
          \end{itemize}
          Then
          \begin{itemize}
            \item $\forall x \in X, \min\set*{{a_{k}, a_{n + 1}}} \leq x$
            \item $\forall x \in X, x \leq \max\set*{{a_{K}, a_{n + 1}}}$
          \end{itemize}
          Therefore, $P(n + 1)$ is true.
  \end{itemize}
  By \cref{an:def:mathematical-induction},
  \[
    \forall n \in \N, P(n) \text{ is true}
  \]
\end{proof}

\begin{lemma} \label{an:lem:properties-of-Nm-Nn}
  Suppose $n, m \in [0, \infty)_{\N}$. Then
  \begin{enumerate}[label=\textbf{(\Alph*)}]
    \item $\paren*{\exists f: [1, n]_{\N} \hookrightarrow [1, m]_{\N}} \implies n \leq m$
    \item $\paren*{\exists f: [1, n]_{\N} \twoheadrightarrow [1, m]_{\N}} \implies n \geq m \geq 1 \lor n = m = 0$
    \item $\paren*{\exists f: [1, n]_{\N} \xrightarrow{\sim} [1, m]_{\N}} \implies n = m$
  \end{enumerate}
\end{lemma}
