\begin{proof}
  Suppose $I$ is a $k$-cell in $\R^{k}$. Define
  \[
    I_{1} \coloneqq I = \prod_{j = 1}^{k} \bracket*{a_{1, j}, b_{1, j}} \subseteq \R^{k}
  \]
  where
  \[
    \forall j \in [1, k]_{\N}, - \infty < a_{1, j} \leq b_{1, j} < + \infty
  \]
  Define
  \[
    \forall j \in [1, k]_{\N}, \forall m \in [1, 2]_{\N}, I_{1, j, m} \coloneqq \begin{dcases}
      \bracket*{a_{1, j}, \frac{a_{1, j} + b_{1, j}}{2}}, & \text{if } m = 1 \\
      \bracket*{\frac{a_{1, j} + b_{1, j}}{2}, b_{1, j}}, & \text{if } m = 2
    \end{dcases}
  \]
  Define
  \[
    S_{k} \coloneqq \prod_{j = 1}^{k} \set*{1, 2} = \setbuilder*{\bm{m} \in \R^{k}}{\forall j \in [1, k]_{\N}, m_{j} \in [1, 2]_{\N}}
  \]
  Then
  \[
    I_{1} = \bigcup_{\bm{m} \in S_{k}} \paren*{\prod_{j = 1}^{k} I_{1, j, m_{j}}}
  \]
  By \cref{an:thm:product-of-finite-sets-is-finite},
  \[
    \abs*{S_{k}} < \aleph_{0}
  \]
  Suppose $\set*{G_{\alpha}}_{\alpha \in A}$ satisfies
  \begin{itemize}
    \item $\forall \alpha \in A, G_{\alpha}$ is open in $\paren*{\R^{k}, \norm*{\cdot}}$
    \item $I_{1} \subseteq \bigcup_{\alpha \in A} G_{\alpha}$
  \end{itemize}
  Assume
  \[
    \nexists B \subseteq A, \bracket*{\begin{gathered}
        \abs*{B} < \aleph_{0} \\
        I_{1} \subseteq \bigcup_{\alpha \in B} G_{\alpha}
      \end{gathered}}
  \]
  Then
  \[
    \begin{array}{c}
      \forall \bm{m} \in S_{k}, \prod_{j = 1}^{k} I_{1, j, m_{j}}
      \subseteq I_{1} \subseteq \bigcup_{\alpha \in A} G_{\alpha} \\
      \Downarrow                                                  \\
      \set*{G_{\alpha}}_{\alpha \in A} \text{ is an open cover of } \prod_{j = 1}^{k} I_{1, j, m_{j}}
    \end{array}
  \]
  Assume
  \[
    \forall \bm{m} \in S_{k}, \exists A_{\bm{m}} \subseteq A, \bracket*{\begin{gathered}
        \abs*{A_{\bm{m}}} < \aleph_{0} \\
        \prod_{j = 1}^{k} I_{1, j, m_{j}} \subseteq \bigcup_{\alpha \in A_{\bm{m}}} G_{\alpha}
      \end{gathered}}
  \]
  By \cref{an:thm:arbitrary-union-of-open-sets-is-open},
  \[
    \bigcup_{\alpha \in A_{\bm{m}}} G_{\alpha} \text{ is open}
  \]
  Then
  \[
    I_{1} \subseteq \bigcup_{\bm{m} \in S_{k}} \paren*{\bigcup_{\alpha \in A_{\bm{m}}} G_{\alpha}}
    = \bigcup_{\alpha \in A} G_{\alpha}
  \]
  where
  \[
    A \coloneqq \bigcup_{\bm{m} \in S_{k}} A_{\bm{m}}
  \]
  By \cref{an:thm:finite-union-of-finite-sets-is-finite},
  \[
    \begin{array}{c}
      \abs*{A} < \aleph_{0} \\
      \Downarrow            \\
      \bot
    \end{array}
  \]
  Therefore,
  \[
    \exists \bm{m_{1}} \in S_{k}, \nexists A_{\bm{m_{1}}} \subseteq A, \bracket*{\begin{gathered}
        \abs*{A_{\bm{m_{1}}}} < \aleph_{0} \\
        \prod_{j = 1}^{k} I_{1, j, \paren*{m_{1}}_{j}} \subseteq \bigcup_{\alpha \in A_{\bm{m_{1}}}} G_{\alpha}
      \end{gathered}}
  \]
  Define
  \[
    I_{2} \coloneqq \prod_{j = 1}^{k} \bracket{a_{2, j}, b_{2, j}} \subseteq \R^{k}
  \]
  where $a_{2, j}, b_{2, j} \in \R$ is defined as
  \begin{itemize}
    \item $a_{2, j} \coloneqq \begin{dcases}
              a_{1, j},                      & \text{if } \paren*{m_{1}}_{j} = 1 \\
              \frac{a_{1, j} + b_{1, j}}{2}, & \text{if } \paren*{m_{1}}_{j} = 2
            \end{dcases}$
    \item $b_{2, j} \coloneqq \begin{dcases}
              \frac{a_{1, j} + b_{1, j}}{2}, & \text{if } \paren*{m_{1}}_{j} = 1 \\
              b_{1, j},                      & \text{if } \paren*{m_{1}}_{j} = 2
            \end{dcases}$
  \end{itemize}
  for all $j \in [1, k]_{\N}$. Then
  \[
    \begin{array}{c}
      \forall j \in [1, k]_{\N}, \bracket*{a_{2, j}, b_{2, j}} =
      I_{1, j, \paren*{m_{1}}_{j}} \subseteq \bracket*{a_{1, j}, b_{1, j}} \\
      \Downarrow                                                           \\
      I_{2} = \prod_{j = 1}^{k} I_{1, j, \paren*{m_{1}}_{j}} \subseteq I_{1}
    \end{array}
  \]
  Define
  \[
    I_{2, j, m} \coloneqq \begin{dcases}
      \bracket*{a_{2, j}, \frac{a_{2, j} + b_{2, j}}{2}}, & \text{if } m = 1 \\
      \bracket*{\frac{a_{2, j} + b_{2, j}}{2}, b_{2, j}}, & \text{if } m = 2
    \end{dcases}
  \]
  for all $j \in [1, k]_{\N}$ and $m \in [1, 2]_{\N}$. Then
  \[
    I_{2} = \bigcup_{\bm{m} \in S_{k}} \paren*{\prod_{j = 1}^{k} I_{2, j, m_{j}}}
  \]
  We have shown that
  \begin{itemize}
    \item $\set*{G_{\alpha}}_{\alpha \in A}$ is an open cover of $I_{2}$
    \item $I_{2}$ has no finite subcover
  \end{itemize}
  Then
  \[
    \begin{array}{c}
      \forall \bm{m} \in S_{k}, \prod_{j = 1}^{k} I_{2, j, m_{j}}
      \subseteq I_{2} \subseteq \bigcup_{\alpha \in A} G_{\alpha} \\
      \Downarrow                                                  \\
      \set*{G_{\alpha}}_{\alpha \in A} \text{ is an open cover of } \prod_{j = 1}^{k} I_{2, j, m_{j}}
    \end{array}
  \]
  \[
    \prod_{j = 1}^{k} I_{2, j, m_{j}} \subseteq I_{2} \subseteq \bigcup_{\alpha \in A} G_{\alpha},
  \]
  Therefore,
  \[
    \exists \bm{m_{2}} \in S_{k},\prod_{j = 1}^{k} I_{2, j, \paren*{m_{2}}_{j}} \text{ has no finite subcover}
  \]
  By repeating this process, we can construct the following sequences:
  \begin{itemize}
    \item The sequence $\paren*{\bm{m}_{n}}_{n = 1}^{\infty}$ in $S_{k}$.
    \item The sequences $\paren*{a_{n, j}}_{n = 1}^{\infty}$ and
          $\paren*{b_{n, j}}_{n = 1}^{\infty}$ in $\R$ for all $j = 1, 2, \dots, k$.
    \item The sequence $\paren*{I_{n, j, m}}_{n = 1}^{\infty}$ of intervals in $\R$
          for all $j = 1, 2, \dots, k$ and $m = 1, 2$.
    \item The sequence $\paren*{I_{n}}_{n = 1}^{\infty}$ of $k$-cells in $\R^{k}$.
  \end{itemize}
  We have the following properties:
  \begin{itemize}
    \item $I_{n + 1} \subseteq I_{n}$ for all $n \in \N$.
    \item $I_{n}$ has no finite subcover for all $n \in \N$.
    \item $a_{n, j} \leq a_{n + 1, j} \leq b_{n + 1, j} \leq b_{n, j}$ for all $j = 1, 2, \dots, k$ and $n \in \N$.
    \item $b_{n + 1, j} - a_{n + 1, j} = \frac{1}{2}(b_{n, j} - a_{n, j}) \implies b_{n, j} - a_{n, j} =
            \frac{1}{2^{n - 1}}(b_{1, j} - a_{1, j})$ for all $j = 1, 2, \dots, k$ and $n \in \N$.
  \end{itemize}
  Let $\delta \coloneqq \sqrt{\sum_{j = 1}^{k} (b_{1, j} - a_{1, j})^{2}}$. Then we have
  \[
    \norm*{\bm{x} - \bm{y}} \leq \sqrt{\sum_{j = 1}^{k} (b_{n, j} - a_{n, j})^{2}} =
    \frac{\delta}{2^{n - 1}} \quad \text{for all } \bm{x}, \bm{y} \in I_{n} \text{ and } n \in \N.
  \]
  By \cref{an:thm:nested-k-cells}, we have $\bigcap_{n = 1}^{\infty} I_{n} \neq \vnot$.
  Suppose $\bm{\beta} \in \bigcap_{n = 1}^{\infty} I_{n}$. Then there exists $\alpha_{0} \in A$
  such that $\bm{\beta} \in G_{\alpha_{0}}$. Since $G_{\alpha_{0}}$ is open, there exists $\veps \in \R^{+}$ such that
  \[
    B_{\veps}(\bm{\beta}) \subseteq G_{\alpha_{0}}.
  \]
  By \cref{an:thm:archimedean-property-corollary}, there exists $N \in \N$
  such that $\frac{1}{2^{N}} < \frac{\veps}{2\delta}$. Since $\bm{\beta} \in I_{N}$, we have
  \[
    \norm*{\bm{\beta} - \bm{\gamma}} \leq \frac{\delta}{2^{N - 1}} < \veps \quad \text{for all } \bm{\gamma} \in I_{N},
  \]
  which implies that $I_{N} \subseteq B_{\veps}(\bm{\beta}) \subseteq G_{\alpha_{0}}$. Therefore, $I_{N}$ has a
  finite subcover $\set*{G_{\alpha_{0}}}$, which is a contradiction. Thus $I$ has a finite subcover, and $I$ is compact.
\end{proof}

\begin{corollary} \label{an:cor:closed-interval-in-R-is-compact}
  Suppose $- \infty < a \leq b < + \infty$. Then
  \[
    [a, b] \text{ is compact in } \paren*{\R, \norm*{\cdot}}
  \]
\end{corollary}

\begin{theorem} \label{an:thm:heine-borel-theorem}
  Suppose $n \in \N$ and $E \subseteq \R^{n}$.
  Then we have
  \[
    E \text{ is compact} \iff E \text{ is closed and bounded}.
  \]
  This theorem is called the \textbf{Heine-Borel Theorem}.
\end{theorem}

\begin{proof}
  \textbf{($\implies$)} Suppose $E$ is compact. $E$ is closed by \cref{an:thm:compact-set-is-closed}, and $E$ is
  bounded by \cref{an:thm:compact-set-is-bounded}.
  \\[6mm]
  \textbf{($\impliedby$)} Suppose $E$ is closed and bounded. Then there exists some $\bm{p} \in \R^{n}$ and
  $r \in \R^{+}$ such that $E \subseteq B_{r}(\bm{p})$. Let $I$ be a $n$-cell in $\R^{n}$ defined as
  \[
    I \coloneqq \prod_{j = 1}^{n} \bracket*{p_{j} - r, p_{j} + r} \subseteq \R^{n}.
  \]
  Then we have $E \subseteq B_{r}(\bm{p}) \subseteq I$. By \cref{an:thm:k-cell-is-compact}, $I$ is compact.
  Since $E$ is closed, by \cref{an:thm:closed-subset-of-compact-set-is-compact}, $E$ is compact.
\end{proof}
