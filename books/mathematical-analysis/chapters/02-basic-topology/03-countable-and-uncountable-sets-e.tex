\bigskip

\begin{theorem} \label{an:thm:R-is-uncountable}
  \[
    \abs*{\R} > \aleph_{0}
  \]
\end{theorem}

\begin{proof}
  Consider
  \[
    \begin{array}{c}
      \N \subseteq \R                                                                     \\
      \by{\Cref{an:lem:N-is-infinite,an:cor:infinite-subset-implies-infinite}} \Downarrow \\
      \abs*{\R} \geq \aleph_{0}
    \end{array}
  \]
  Define $f: \N \hookrightarrow [0, 1)$ by
  \[
    \forall n \in \N, f(n) = \frac{1}{n + 1}
  \]
  By \cref{an:lem:N-is-infinite,an:thm:injection-from-infinite-set-implies-infinite},
  \[
    \abs*{[0, 1)} \geq \aleph_{0}
  \]
  Define $g: [0, 1) \to \prod_{n = 1}^{\infty} \set*{0, 1}$ by
  \[
    \forall x \in [0, 1), g(x) = \GBE_{2}(x)
  \]
  By \cref{an:thm:same-greedy-base-expansion-implies-equality},
  \[
    g: [0, 1) \hookrightarrow \prod_{n = 1}^{\infty} \set*{0, 1}
  \]
  Define
  \[
    Y \coloneqq g([0, 1)) \subseteq \prod_{n = 1}^{\infty} \set*{0, 1}
  \]
  Then
  \[
    g|^{Y}: [0, 1) \xrightarrow{\sim} Y
  \]
  Assume $\abs*{[0, 1)} = \aleph_{0}$. By \cref{an:thm:bijection-equivalence-relation},
  \[
    \abs*{Y} = \aleph_{0}
  \]
  Define
  \[
    Z \coloneqq \prod_{n = 1}^{\infty} \set*{0, 1} \setminus Y
  \]
  By
  \Cref{an:thm:no-real-number-matching-greedy-base-expansion,an:thm:no-real-number-matching-greedy-base-expansion-converse},
  \[
    Z = \setbuilder*{\paren*{a_{n}}_{n = 1}^{\infty} \in \prod_{n = 1}^{\infty}
    \set*{0, 1}}{\exists N \in \N, \forall n \in [N, \infty)_{\N}, a_{n} = 1}
  \]
  Define
  \[
    \forall N \in \N, Z_{N} \coloneqq \setbuilder*{\paren*{a_{n}}_{n = 1}^{\infty} \in
    \prod_{n = 1}^{\infty} \set*{0, 1}}{\forall n \in [N, \infty)_{\N}, a_{n} = 1} \neq \vnot
  \]
  Then
  \begin{itemize}
    \item If $N = 1$, then
          \[
            \begin{array}{c}
              Z_{1} = \set*{\paren*{1}_{n = 1}^{\infty}} \\
              \Downarrow                                 \\
              \abs*{Z_{1}} = 1 < \aleph_{0}
            \end{array}
          \]
    \item Otherwise, define $F_{N}: \prod_{n = 1}^{N - 1} \set*{0, 1} \xrightarrow{\sim} Z_{N}$ as
          \[
            \forall \paren*{a_{n}}_{n = 1}^{N - 1} \in \prod_{n = 1}^{N - 1} \set*{0, 1},
            F_{N}\paren*{\paren*{a_{n}}_{n = 1}^{N - 1}} = \paren*{a_{n}}_{n = 1}^{N - 1} + \paren*{1}_{n = N}^{\infty}
          \]
          By \cref{an:thm:product-of-finite-sets-is-finite},
          \[
            \begin{array}{c}
              \forall N \in \N, \abs*{\prod_{n = 1}^{N - 1} \set*{0, 1}} < \aleph_{0} \\
              \by{\Cref{an:thm:surjection-from-finite-set-implies-finite}} \Downarrow \\
              \forall N \in [2, \infty)_{\N}, \abs*{Z_{N}} < \aleph_{0}
            \end{array}
          \]
  \end{itemize}
  Therefore,
  \[
    \begin{array}{c}
      Z = \bigcup_{N = 1}^{\infty} Z_{N}                                                \\
      \by{\Cref{an:thm:countable-union-of-finite-sets-is-at-most-countable}} \Downarrow \\
      \abs*{Z} \leq \aleph_{0}
    \end{array}
  \]
  By \cref{an:cor:finite-union-of-countable-sets-is-countable},
  \[
    \begin{array}{c}
      \prod_{n = 1}^{\infty} \set*{0, 1} = Y \cup Z                                \\
      \Downarrow                                                                   \\
      \abs*{\prod_{n = 1}^{\infty} \set*{0, 1}} = \aleph_{0}                       \\
      \by{\Cref{an:lem:infinite-product-of-finite-sets-is-uncountable}} \Downarrow \\
      \bot
    \end{array}
  \]
  Therefore,
  \[
    \begin{array}{c}
      \abs*{[0, 1)} > \aleph_{0} \\
      \Downarrow                 \\
      \abs*{\R} > \aleph_{0}
    \end{array}
  \]
\end{proof}
