\begin{proof}
  Suppose
  \begin{itemize}
    \item $\abs*{X} < \aleph_{0}$
    \item $\abs*{Y} < \aleph_{0}$
  \end{itemize}
  Then
  \begin{itemize}
    \item $\exists n \in [0, \infty)_{\N}, \exists f: [1, n]_{\N} \xrightarrow{\sim} X$
    \item $\exists m \in [0, \infty)_{\N}, \exists g: [1, m]_{\N} \xrightarrow{\sim} Y$
  \end{itemize}
  Therefore,
  \begin{itemize}
    \item If $n = 0$ or $m = 0$, then
          \[
            \begin{array}{c}
              X \times Y = \varnothing \\
              \Downarrow               \\
              \abs*{X \times Y} = 0 < \aleph_{0}
            \end{array}
          \]
    \item Otherwise, define $h: [1, nm]_{\N} \to X \times Y$ as
          \[
            \forall k \in [1, nm]_{\N}, h(k) = \paren*{f\paren*{\ceil*{\frac{k}{m}}}, g\paren*{k - m\paren*{\ceil*{\frac{k}{m}} - 1}}}
          \]
          Suppose
          \begin{itemize}
            \item $k \in [1, nm]_{\N}$
            \item $j_{k} = \ceil*{\frac{k}{m}} \in \Z$
          \end{itemize}
          Then
          \begin{itemize}
            \item Since $k \in [1, nm]_{\N}$,
                  \[
                    \begin{array}{c}
                      1 \leq k \leq nm                  \\
                      \Downarrow                        \\
                      1 \leq \ceil*{\frac{k}{m}} \leq n \\
                      \Downarrow                        \\
                      j_{k} \in [1, n]_{\N}
                    \end{array}
                  \]
            \item By \cref{an:def:floor-ceiling-functions},
                  \[
                    \begin{array}{c}
                      j_{k} = \min\setbuilder*{i \in \Z}{i \geq \frac{k}{m}} \\
                      \Downarrow                                             \\
                      j_{k} - 1 < \frac{k}{m} \leq j_{k}                     \\
                      \Downarrow                                             \\
                      0 < k - m(j_{k} - 1) \leq m                            \\
                      \Downarrow                                             \\
                      k - m(j_{k} - 1) \in [1, m]_{\N}
                    \end{array}
                  \]
          \end{itemize}
          Therefore, $h$ is well-defined. Then
          \begin{itemize}
            \item \textbf{Injectivity}: Suppose
                  \begin{itemize}
                    \item $k_{1}, k_{2} \in [1, nm]_{\N}$
                    \item $h(k_{1}) = h(k_{2})$
                  \end{itemize}
                  Then
                  \[
                    \begin{array}{c}
                      f\paren*{j_{k_{1}}} = f\paren*{j_{k_{2}}}                        \\
                      \by{\ensuremath{f: [1, n]_{\N} \xrightarrow{\sim} X}} \Downarrow \\
                      j_{k_{1}} = j_{k_{2}}
                    \end{array}
                  \]
                  Therefore,
                  \[
                    \begin{array}{c}
                      g\paren*{k_{1} - m(j_{k_{1}} - 1)} = g\paren*{k_{2} - m(j_{k_{2}} - 1)} \\
                      \by{\ensuremath{g: [1, m]_{\N} \xrightarrow{\sim} Y}} \Downarrow        \\
                      k_{1} - m(j_{k_{1}} - 1) = k_{2} - m(j_{k_{2}} - 1)                     \\
                      \by{\ensuremath{j_{k_{1}} = j_{k_{2}}}} \Downarrow                      \\
                      k_{1} = k_{2}
                    \end{array}
                  \]
            \item \textbf{Surjectivity}: Suppose $(x, y) \in X \times Y$. Then
                  \begin{itemize}
                    \item Since $(x, y) \in X \times Y$,
                          \[
                            \begin{array}{c}
                              x \in X                                                          \\
                              \by{\ensuremath{f: [1, n]_{\N} \xrightarrow{\sim} X}} \Downarrow \\
                              \exists j \in [1, n]_{\N}, f(j) = x
                            \end{array}
                          \]
                    \item Since $(x, y) \in X \times Y$,
                          \[
                            \begin{array}{c}
                              y \in Y                                                          \\
                              \by{\ensuremath{g: [1, m]_{\N} \xrightarrow{\sim} Y}} \Downarrow \\
                              \exists i \in [1, m]_{\N}, g(i) = y
                            \end{array}
                          \]
                  \end{itemize}
                  Let
                  \[
                    k \coloneqq m(j - 1) + i \in [1, nm]_{\N}
                  \]
                  Then
                  \[
                    h(k) = \paren*{f(j), g(i)} = (x, y)
                  \]
          \end{itemize}
          Therefore,
          \[
            \begin{array}{c}
              h: [1, nm]_{\N} \xrightarrow{\sim} X \times Y \\
              \Downarrow                                    \\
              \abs*{X \times Y} = nm < \aleph_{0}
            \end{array}
          \]
  \end{itemize}
\end{proof}

\begin{theorem} \label{an:thm:product-of-finite-sets-is-finite}
  \[
    \forall n \in \N, \forall X_{1}, X_{2}, \ldots, X_{n},
    \paren*{\forall k \in [1, n]_{\N}, \abs*{X_{k}} < \aleph_{0}}
    \implies \abs*{\prod_{k = 1}^{n} X_{k}} < \aleph_{0}
  \]
\end{theorem}

\begin{proof}
  Define
  \[
    \forall n \in \N, P(n): \paren*{\forall k \in [1, n]_{\N}, \abs*{X_{k}} < \aleph_{0}}
    \implies \abs*{\prod_{k = 1}^{n} X_{k}} < \aleph_{0}
  \]
  Then
  \begin{itemize}
    \item \textbf{Base case:} Suppose
          \begin{itemize}
            \item $\abs*{X_{1}} < \aleph_{0}$
          \end{itemize}
          Then
          \[
            \begin{array}{c}
              \abs*{X_{1}} < \aleph_{0}                   \\
              \Downarrow                                  \\
              \abs*{\prod_{k = 1}^{1} X_{k}} < \aleph_{0} \\
              \Downarrow                                  \\
              P(1) \text{ is true}
            \end{array}
          \]
    \item \textbf{Inductive step:} Suppose $P(n)$ is true. Suppose
          \begin{itemize}
            \item $\forall k \in [1, n + 1]_{\N}, \abs*{X_{k}} < \aleph_{0}$
          \end{itemize}
          Since $P(n)$ is true,
          \[
            \abs*{\prod_{k = 1}^{n} X_{k}} < \aleph_{0}
          \]
          Then
          \[
            \begin{array}{c}
              \prod_{k = 1}^{n + 1} X_{k} \sim \paren*{\prod_{k = 1}^{n} X_{k}} \times X_{n + 1} \\
              \by{\Cref{an:lem:product-of-two-finite-sets-is-finite}} \Downarrow                 \\
              \abs*{\prod_{k = 1}^{n + 1} X_{k}} < \aleph_{0}                                    \\
              \Downarrow                                                                         \\
              P(n + 1) \text{ is true}
            \end{array}
          \]
  \end{itemize}
  By \cref{an:thm:mathematical-induction},
  \[
    \forall n \in \N, P(n) \text{ is true}
  \]
\end{proof}

\begin{theorem} \label{an:thm:cantors-theorem}
  \[
    \forall X, \nexists f: X \twoheadrightarrow \mathcal{P}(X)
  \]
  This theorem is called \textbf{Cantor's theorem}.
\end{theorem}

\begin{proof}
  Suppose
  \[
    \exists f: X \twoheadrightarrow \mathcal{P}(X)
  \]
  Define
  \[
    S \coloneqq \setbuilder*{x \in X}{x \notin f(x)} \subseteq X
  \]
  Then
  \[
    \begin{array}{c}
      S \subseteq X        \\
      \Downarrow           \\
      S \in \mathcal{P}(X) \\
      \Downarrow           \\
      \exists x \in X, f(x) = S
    \end{array}
  \]
  Consider
  \begin{itemize}
    \item If $x \in S$, then
          \[
            \begin{array}{c}
              x \notin f(x) = S \\
              \Downarrow        \\
              \bot
            \end{array}
          \]
    \item If $x \notin S$, then
          \[
            \begin{array}{c}
              x \in f(x) = S \\
              \Downarrow     \\
              \bot
            \end{array}
          \]
  \end{itemize}
  Therefore,
  \[
    \nexists f: X \twoheadrightarrow \mathcal{P}(X)
  \]
\end{proof}
